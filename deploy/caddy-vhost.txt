# Vhost walidatora dla Caddy — do dopisania do Caddyfile na serwerze.
# Po zmianie przeładuj Caddy (np. `caddy reload` w jego kontenerze).
#
# ⚠ TEN HOST JEST CELOWO OTWARTY — bez Basic Auth. To narzędzie publiczne i nie POKAZUJE żadnych
# cudzych danych. Zapisuje historię sprawdzeń (domyślnie z treścią pliku) do SQLite na serwerze,
# ale żadna publiczna trasa jej nie oddaje: `/api/historia` istnieje tylko z `HISTORIA_TOKEN`.
# Gdy którakolwiek trasa zacznie oddawać CUDZE dane, host idzie za bramę logowania.

valid.example.com {
	encode zstd gzip
	# ⚠ NIE 127.0.0.1, jeśli Caddy sam stoi w kontenerze: 127.0.0.1 to wtedy on sam (502).
	# Oba kontenery stoją w jednej sieci Dockera (`SIEC` w deploy/deploy.env) — adresujemy
	# po nazwie kontenera i porcie WEWNĘTRZNYM.
	reverse_proxy walidator-prod:8139
	header {
		Strict-Transport-Security "max-age=31536000; includeSubDomains"
		X-Content-Type-Options "nosniff"
		Referrer-Policy "no-referrer"
		# ⚠ Strona nie ładuje ANI JEDNEGO zasobu z zewnątrz (style i skrypt są w HTML-u),
		# więc CSP może być maksymalnie wąskie. Gdy ktoś dołoży CDN, to się zepsuje
		# GŁOŚNO — i o to chodzi.
		Content-Security-Policy "default-src 'none'; style-src 'unsafe-inline'; script-src 'unsafe-inline'; connect-src 'self'; img-src 'self' data:; form-action 'none'; base-uri 'none'; frame-ancestors 'none'"
	}
}

# ⚠ BEZ STAGINGU — jedna publiczna instancja. Nie dopisuj tu drugiego hosta.
